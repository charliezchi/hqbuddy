> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `-flow -bin_only <name>` mode to `hqbuddy` that trims the full `-flow` TCL to keep only synthesis → pack → place → route → bitgen, suppressing all intermediate file writes, and lets the user name the final bitstream.

**Architecture:** Reuse the existing `run_flow()` path to generate the full TCL via `hqprj2tcl`, then post-process the generated file with line filters (removing `nl.write`, `nl.report`, `xpn.write`, `ta.fmax.report`) and replace the `design.bitgen` target with the user-supplied name plus the original extension (`bit` or `bin`).

**Tech Stack:** Python 3.12, PyInstaller for `hqbuddy.exe`, existing `hqbuddy` modules (`flow.py`, `__main__.py`).

## Global Constraints

- Version target: `2.2.4`
- Keep changes minimal and consistent with existing `run_flow_looptdo()` pattern in `hqbuddy/flow.py`
- All file paths written to `.hqprj` or TCL must use forward slashes `/`
- `-bin_only` must be recognized in `cmd_flow()` after `-flow`
- Help text in `show_help()` must be updated under the `-flow` option

```python
def _make_bin_only_tcl(full_tcl_path: str, bin_name: str) -> None:
    """
    Post-process the generated TCL for bin-only mode.

    Keeps synthesis through bitgen, but removes intermediate file writes
    (netlist dumps, XPN files, timing reports). The final design.bitgen
    target is replaced with <bin_name>.<original_ext>.

    Args:
        full_tcl_path: Path to the TCL generated by hqprj2tcl.
        bin_name: User-specified base name for the final bitstream.
    """
    with open(full_tcl_path, "r", encoding="utf-8") as f:
        lines = f.readlines()

    # Commands that write intermediate files we want to drop
    drop_prefixes = (
        "nl.write",
        "nl.report",
        "xpn.write",
        "ta.fmax.report",
    )

    kept = []
    bitgen_ext = None
    bitgen_idx = None

    for i, line in enumerate(lines):
        stripped = line.strip()

        # Drop intermediate file output commands
        if any(stripped.startswith(p) for p in drop_prefixes):
            continue

        # Locate design.bitgen and extract original extension
        if stripped.startswith("design.bitgen"):
            parts = stripped.split()
            if len(parts) >= 2:
                original_target = parts[1]
                _, ext = os.path.splitext(original_target)
                if ext:
                    bitgen_ext = ext  # e.g. ".bit" or ".bin"
            bitgen_idx = len(kept)
            kept.append(line)
            continue

        kept.append(line)

    if bitgen_idx is None:
        print("Error: could not locate design.bitgen command in generated TCL")
        sys.exit(1)

    if not bitgen_ext:
        print("Warning: could not detect bitgen extension; defaulting to .bit")
        bitgen_ext = ".bit"

    # Normalize user-provided name: strip known extension if it matches,
    # then append the original script extension
    clean_name = bin_name
    if clean_name.lower().endswith(bitgen_ext.lower()):
        clean_name = clean_name[:-len(bitgen_ext)]
    final_target = f"{clean_name}{bitgen_ext}"

    # Rewrite design.bitgen line, preserving any extra arguments
    original_line = kept[bitgen_idx]
    prefix = original_line[:original_line.find("design.bitgen") + len("design.bitgen")]
    kept[bitgen_idx] = f"{prefix} {final_target}\n"

    with open(full_tcl_path, "w", encoding="utf-8") as f:
        f.writelines(kept)

    print(f"Bin-only mode: intermediate file writes removed; bitgen target -> {final_target}")
```

def run_flow_bin_only(hqprj_path: str, bin_name: str, output_tcl: str | None = None) -> None:
    """
    Run the flow in bin-only mode.

    Generates the full TCL via hqprj2tcl, then trims it to remove all
    intermediate file writes and sets the final bitgen target to bin_name.

    Args:
        hqprj_path: Path to the .hqprj file.
        bin_name: User-specified base name for the final bitstream.
        output_tcl: Optional custom output TCL file name.
    """
    if not os.path.isfile(hqprj_path):
        print(f"Error: file not found: {hqprj_path}")
        sys.exit(1)

    work_dir = os.path.dirname(os.path.abspath(hqprj_path))
    hqfpga_path = _resolve_hqfpga()

    # Generate the full TCL via hqprj2tcl first
    temp_tcl = _generate_temp_tcl(hqprj_path, output_tcl)

    try:
        print(f"Generated temp TCL: {temp_tcl}")

        # Run hqprj2tcl to generate the full flow TCL
        _run_hqfpga(hqfpga_path, temp_tcl, work_dir)

        # Determine the output TCL path (the one hqprj2tcl wrote to)
        if output_tcl:
            flow_tcl = os.path.join(work_dir, output_tcl)
        else:
            flow_tcl = os.path.join(work_dir, "run_hqprj.tcl")

        if not os.path.isfile(flow_tcl):
            print(f"Error: hqprj2tcl did not generate expected TCL: {flow_tcl}")
            sys.exit(1)

        # Post-process for bin-only mode
        _make_bin_only_tcl(flow_tcl, bin_name)
        print(f"Bin-only TCL ready: {os.path.abspath(flow_tcl)}")

    finally:
        if os.path.exists(temp_tcl):
            os.remove(temp_tcl)
            print(f"")
            print(f"Cleaned up temp TCL: {temp_tcl}")
```

def cmd_flow(args):
    """Run hqprj2tcl flow via hqfpga."""
    # Check for looptdo mode
    looptdo = False
    if args and args[0] == '-looptdo':
        looptdo = True
        args = args[1:]

    # Check for bin-only mode
    bin_only_name = None
    if '-bin_only' in args:
        idx = args.index('-bin_only')
        if idx + 1 >= len(args):
            print("Error: -bin_only requires a bin name")
            sys.exit(1)
        bin_only_name = args[idx + 1]
        args = args[:idx] + args[idx + 2:]

    if looptdo and bin_only_name:
        print("Error: -looptdo and -bin_only cannot be used together")
        sys.exit(1)

    hqprj_path, output_tcl = _parse_args_with_output(args)

    if looptdo:
        run_flow_looptdo(hqprj_path, output_tcl)
    elif bin_only_name:
        run_flow_bin_only(hqprj_path, bin_only_name, output_tcl)
    else:
        run_flow(hqprj_path, output_tcl)
```
